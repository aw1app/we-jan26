## One more SP
DELIMITER //
CREATE PROCEDURE CountProductsByPrefix(
    IN productPrefix VARCHAR(100)
)
BEGIN
    SELECT COUNT(*) AS totalProducts
    FROM products
    WHERE name LIKE CONCAT(productPrefix, '%');
END //
DELIMITER ;

// SP
DELIMITER //

CREATE PROCEDURE FindLargestNumber(
    IN num1 INT,
    IN num2 INT
)
BEGIN
    DECLARE largest INT;

    SET largest = num1;

    IF num2 > largest THEN
        SET largest = num2;
    END IF;

    SELECT largest AS LargestNumber;
END //

DELIMITER ;
